public interface IPriceFeed
{
    decimal GetPrice(string sku);
}

Build → MockLite generates MockPriceFeed with setup and verification members for your contract.

02 Configure and verify with the generated API
var prices = new MockPriceFeed();

prices.SetupGetPrice(sku => sku == "SKU-42")
    .Callback(sku => Console.WriteLine(sku))
    .Returns(19.95m);

// sku is inferred as string in both lambdas.
IPriceFeed feed = prices;
decimal price = feed.GetPrice("SKU-42");

prices.VerifyGetPrice(
    sku => sku == "SKU-42", Times.Once);
